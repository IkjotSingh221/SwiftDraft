%% next to the rendered .tex -- everything else in this template ($body$,
%% citeproc-generated references, title block) still applies unchanged.
%%
%% Phase 6 invocation (see templates/ieee_report.yaml for the full command):
%%   pandoc assembled.md --defaults=templates/ieee_report.yaml \
%%     --metadata-file=run_metadata.yaml \
%%     --bibliography=run_bibliography.json -o draft.pdf
%%
